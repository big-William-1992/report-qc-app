# 星衍放射质控软件 · 跨平台依赖（macOS / Linux / Windows 均可干净安装）
#
# 架构说明：现版本为「FastAPI 单服务同源托管 SPA」+「pywebview 桌面壳」。
#   - server/main.py  提供 /api/v1/* REST 接口，并以 StaticFiles 托管 web/static/ 前端
#   - desktop_app.py  后台起 uvicorn，再用系统原生 WebView 加载同一套 SPA（桌面≈浏览器）
#   - src/app.py      旧 Tkinter 单体版，保留作历史参考（不再是主入口）
#
# ── Web 服务与桌面壳（主程序运行必需）───────────────────────────────
fastapi>=0.110                # REST 后端框架
uvicorn>=0.27                 # ASGI 服务器（desktop_app.py 以后台线程拉起）
python-multipart>=0.0.9       # UploadFile 表单解析：/ocr/upload、/samples/import/upload 必需
sqlalchemy>=1.4,<3.0          # ORM 数据层（users/departments/samples/queue/settings 持久化，运行必需；漏列会导致冻结 exe 启动即 ModuleNotFoundError）
pywebview>=4.4                # 桌面壳（macOS 用系统 WKWebView，Windows 用 WebView2）
pythonnet==3.1.0; platform_system == "Windows"  # Windows 下 pywebview 6.x 的 winforms/edgechromium 后端强制依赖 pythonnet(clr)；锁定 3.1.0（runtime 目录结构已验证，netfx 模式可加载）
clr-loader==0.3.1; platform_system == "Windows"  # pythonnet 的 .NET 运行时加载器（PyInstaller 打包必须显式收集；锁定版避免 vfx/coreclr 路径差异）

# ── 屏幕区域 OCR（框选 PACS 报告 → 离线识别）──────────────────────
# rapidocr-onnxruntime 的完整运行依赖（pip 会自动解析 Requires，但此处显式列出，

rapidocr-onnxruntime>=1.3.0   # 本地离线中文 OCR（模型已内置在 assets/ocr_models，无需联网）
Pillow>=10.0.0                # 屏幕截图（PIL.ImageGrab）与图像预处理
numpy>=1.24                   # rapidocr / onnxruntime 基础依赖
onnxruntime>=1.16             # OCR 推理引擎后端
opencv-python-headless>=4.5   # rapidocr 图像处理（cv2）。headless 版无 GUI/dll 冗余
pyclipper>=1.3                # rapidocr 文本检测依赖
Shapely>=2.0                  # rapidocr 文本检测依赖
PyYAML>=6.0                   # rapidocr 配置加载
six>=1.16                     # rapidocr 兼容层
cryptography>=41.0            # 离线激活码(Ed25519)校验必需：license_utils 在验证激活码时懒加载，运行期必须使用
